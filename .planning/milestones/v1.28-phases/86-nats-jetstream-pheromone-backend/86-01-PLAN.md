#[async_trait]
pub trait PheromoneSubstrate: Send + Sync {
    async fn deposit(&self, deposit: PheromoneDeposit) -> Result<(), SubstrateError>;
    async fn query_concentration(&self, threat_class: &ThreatClass, now: i64) -> Result<PheromoneConcentration, SubstrateError>;
    async fn query_deposits(&self, query: DepositQuery) -> Result<Vec<PheromoneDeposit>, SubstrateError>;
    async fn recent_deposits(&self, limit: usize) -> Result<Vec<PheromoneDeposit>, SubstrateError>;
    async fn gc_evaporated(&self, now: i64) -> Result<usize, SubstrateError>;
    async fn health(&self) -> Result<SubstrateHealth, SubstrateError>;
}

pub enum ConfiguredPheromoneSubstrate {
    InMemory(InMemoryPheromoneSubstrate),
    LocalJournal(LocalJournalPheromoneSubstrate),
}

pub enum SubstrateError {
    PoisonedLock,
    Read { path: PathBuf, source: std::io::Error },
    Write { path: PathBuf, source: std::io::Error },
    Parse { path: PathBuf, line: usize, source: serde_json::Error },
}

   - `JetStreamPheromoneSubstrate::connect(config: PheromoneConfig, nats_url: &str) -> Result<Self, SubstrateError>` (async)
   - Connects to NATS using `async_nats::connect(nats_url).await`
   - Creates a JetStream context: `async_nats::jetstream::new(client)`
   - Creates or gets KV bucket named `"swarm-pheromone-deposits"` with no TTL (application-managed evaporation). Use `jetstream.create_key_value(async_nats::jetstream::kv::Config { bucket: "swarm-pheromone-deposits".into(), ..Default::default() }).await` or `get_key_value` if it already exists.

   **Key format:** `{threat_class_snake}.{timestamp}.{first_8_chars_of_agent_id}` where threat_class_snake is the serde snake_case name of the ThreatClass variant (serialize ThreatClass to JSON string, strip quotes). This enables prefix scanning by threat class.

   **deposit:**
   - Serialize `PheromoneDeposit` to JSON bytes via `serde_json::to_vec`
   - Compute key from deposit fields using the key format above
   - `kv.put(key, value.into()).await`
   - Map errors to a new `SubstrateError::Nats(String)` variant

   **query_concentration:**
   - Use `kv.keys().await` to get all keys, filter by threat_class prefix
   - For each matching key, `kv.get(key).await`, deserialize to `PheromoneDeposit`
   - Compute concentration using the same `concentration_for` helper already in substrate.rs (make it `pub(crate)`)
   - Collect matching deposits into a Vec, then call `concentration_for`

   - Load all keys, deserialize each deposit
   - For deposits where `is_evaporated(now, config.evaporation_threshold)` is true, call `kv.delete(key).await` (or `kv.purge(key).await` to also remove history)
   - Return count of purged entries

   **health:**
   - Return `SubstrateHealth { backend: "jetstream", durable: true, ready: true/false, details: "NATS KV bucket: swarm-pheromone-deposits", deposit_count }` where ready is based on whether the KV store is accessible (try `kv.keys()` or a lightweight status check)
