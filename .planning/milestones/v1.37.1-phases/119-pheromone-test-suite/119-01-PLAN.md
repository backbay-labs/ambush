pub struct PheromoneDeposit {
    pub indicator: serde_json::Value,
    pub threat_class: ThreatClass,
    pub severity: Severity,
    pub confidence: f64,
    pub timestamp: i64,
    pub decay_half_life: f64,
    pub agent_id: AgentId,
    pub signature: Vec<u8>,
    pub agent_key: Vec<u8>,
}

impl PheromoneDeposit {
    pub fn strength_at(&self, now: i64) -> f64;      // confidence * 0.5^(elapsed/half_life)
    pub fn is_evaporated(&self, now: i64, threshold: f64) -> bool; // strength_at < threshold
}

pub struct PheromoneConcentration {
    pub threat_class: ThreatClass,
    pub total_strength: f64,
    pub distinct_sources: usize,
    pub peak_confidence: f64,
}

pub struct EscalationRecord {
    pub mode: SwarmMode,
    pub threat_class: ThreatClass,
    pub total_strength: f64,
    pub distinct_sources: usize,
    pub peak_confidence: f64,
    pub timestamp: i64,
}

pub struct ThreatClassConfig {
    pub threat_class: ThreatClass,
    pub half_life_secs: f64,
    pub evaporation_threshold: f64,
    pub alert_threshold: f64,
    pub incident_threshold: f64,
}

pub struct ThreatIntelEntry {
    pub indicator_type: ThreatIntelIndicatorType,
    pub value: String,
    pub confidence: f64,
    pub expires_at: i64,
}

pub enum ThreatIntelIndicatorType { IpAddress, Domain, FileHash }
