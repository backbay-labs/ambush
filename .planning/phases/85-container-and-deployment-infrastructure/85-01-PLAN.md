Rewrite the `--serve` block in `swarm_detect.rs` main function to add graceful shutdown and config file watching.

1. **Graceful shutdown**: Replace the bare `axum::serve(listener, router).await?` with:
   ```rust
   let server = axum::serve(listener, detect_http_router(state.clone()));
   let shutdown_signal = async {
       let ctrl_c = tokio::signal::ctrl_c();
       let mut sigterm = tokio::signal::unix::signal(
           tokio::signal::unix::SignalKind::terminate()
       ).expect("failed to register SIGTERM handler");  // OK in main init
       tokio::select! {
           _ = ctrl_c => { println!("swarm-detect: received SIGINT, shutting down"); }
           _ = sigterm.recv() => { println!("swarm-detect: received SIGTERM, shutting down"); }
       }
   };
   server.with_graceful_shutdown(shutdown_signal).await?;
   println!("swarm-detect: shutdown complete");
   ```

   Note: The single `expect` on signal registration during binary init is acceptable since there is no meaningful recovery if the OS cannot register signal handlers. However, if the clippy lint still fires, use a match with a descriptive error return instead.

2. **Config file watching via notify + SIGHUP**: Before starting the server, spawn a background task that watches the config file for changes:
   ```rust
   let state_for_reload = state.clone();
   let config_path_for_watch = cli.config.clone();
   tokio::spawn(async move {
       let mut sighup = tokio::signal::unix::signal(
           tokio::signal::unix::SignalKind::hangup()
       ).expect("failed to register SIGHUP handler");

       // Set up file watcher
       let (tx, mut rx) = tokio::sync::mpsc::channel(8);
       let mut watcher = notify::recommended_watcher(move |res: Result<notify::Event, notify::Error>| {
           if let Ok(event) = res {
               if event.kind.is_modify() {
                   let _ = tx.blocking_send(());
               }
           }
       }).expect("failed to create file watcher");

       use notify::Watcher;
       watcher.watch(
           config_path_for_watch.as_ref(),
           notify::RecursiveMode::NonRecursive,
       ).expect("failed to watch config file");

       loop {
           tokio::select! {
               _ = sighup.recv() => {
                   println!("swarm-detect: received SIGHUP, reloading config");
               }
               _ = rx.recv() => {
                   println!("swarm-detect: config file changed, reloading");
               }
           }
           // Debounce: wait 200ms for rapid successive writes
           tokio::time::sleep(std::time::Duration::from_millis(200)).await;
           // Drain any queued notifications
           while rx.try_recv().is_ok() {}
           match state_for_reload.reload_from_disk() {
               Ok(()) => println!("swarm-detect: config reloaded successfully"),
               Err(e) => eprintln!("swarm-detect: config reload failed: {e}"),
           }
       }
   });
   ```

   The `expect` calls during init before the server starts are acceptable for signal/watcher registration that cannot be meaningfully recovered. If clippy complains, convert to early-return errors from main.
